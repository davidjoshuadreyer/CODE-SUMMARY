#include <cstdlib>
#include <iostream>

using namespace std;

// Function prototypes for forward-referenced functions in this file
void getUserInput (int &p, int &q);
int performAddition (int x, int y);
int performSubtraction (int x, int y);

/*
 * The main() calls functions for user input and computations.
 * It also prints out the results.
 */
int main(void) {
    int a, b;
    
    // Get values from the user
    getUserInput (a, b);
    
    // Do the arithmetic calculations
    int sum = performAddition(a, b);
    int diff = performSubtraction(a, b);
    
    // Output the results
    cout << "The sum of " << a << " and " << b << " is " << sum << endl
         << "and their difference is " << diff << endl;
    
    
    return EXIT_SUCCESS;
}

/**
 * Gets the values for two integers from the user.
 * 
 * @param p A pointer to the first number
 * @param q A pointer to the second number
 */
void getUserInput (int &p, int &q) {
     cout << "Enter the first number: ";
     cin >> p;
     cout << "Enter the second number: ";
     cin >> q;
}

/**
 * Adds two integers.
 *
 * @param x The first integer
 * @param y The second integer
 *
 * @return The sum
 */
int performAddition (int x, int y) {
    return (x + y);
}

/**
 * Subtracts two integers.
 *
 * @param x The first integer
 * @param y The second integer
 *
 * @return The difference
 */
int performSubtraction (int x, int y) {
    return (x - y);
}
