#include <iostream>
#include <stdexcept>
using namespace std;
 
class CircularQueue {
static const int CAPACITY = 4;
int data[CAPACITY]{};
int frontIndex = 0; // Next value to remove.
int rearIndex = 0; // Next free slot to insert into.
int count = 0; // Distinguishes empty from full.
public:
bool empty() const { return ; }
bool full() const { return ; }
void enqueue(int value) {
if (full()) throw runtime_error("full queue");
data[] = value;
rearIndex = ;
; // One more stored value.
}
int front() const {
if (empty()) throw runtime_error("empty queue");
return data[];
}
void dequeue() {
if (empty()) throw runtime_error("empty queue");
frontIndex = ;
; // One fewer stored value.
}
}; // data is a member array, so no manual delete is needed.
 
int main() {
CircularQueue q;
q.enqueue(10); q.enqueue(20);
q.enqueue(30); q.enqueue(40); // Full.
q.(); // Remove 10.
q.dequeue(); // Remove 20.
q.enqueue(50); q.enqueue(60); // Wrap into freed slots.
while (!q.()) {
cout << q.front() << ' ';
q.dequeue();
}
cout << '\n'; // Expected order: 30 40 50 60
}
