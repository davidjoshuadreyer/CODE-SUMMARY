    int isEmpty() { return head == nullptr; }
    void addToHead(int el) {
        head = new IntSLLNode(el, head);
        if (tail == nullptr) tail = head;
    }
    void addToTail(int el) {
        if (tail != nullptr) {
            tail->next = new IntSLLNode(el);
            tail = tail->next;
        } else head = tail = new IntSLLNode(el);

        if (head == tail) head = tail = nullptr;
        else head = head->next;
        delete tmp;
        return el;
    }
    ~IntSLList() { while (!isEmpty()) deleteFromHead(); }

    IntSLList& operator=(const IntSLList&) = delete;

        newNode->next = topOfStack;

        if (isEmpty()) throw StackException("Pop from empty stack");
        T item = topOfStack->theItem;
        StackNode *old = topOfStack;
