Lab 9 — Vector struct with operations (full pattern)
// vectorMath.h
#define MAX_VECTOR_SIZE 20
typedef struct { int size; double data[MAX_VECTOR_SIZE]; } Vector;
Vector *vectorMath(const Vector v1, const Vector v2, Vector *result, char op);
void    printVector(const Vector v, FILE *stream);

// vectorMath.c
Vector *vectorMath(const Vector v1, const Vector v2, Vector *result, char op) {
    if (v1.size != v2.size) return NULL;   // size mismatch
    result->size = v1.size;
    switch (op) {
        case '+': for (int i=0; i<v1.size; i++) result->data[i] = v1.data[i] + v2.data[i]; break;
        case '-': for (int i=0; i<v1.size; i++) result->data[i] = v1.data[i] - v2.data[i]; break;
        case '*': for (int i=0; i<v1.size; i++) result->data[i] = v1.data[i] * v2.data[i]; break;
        case '/': for (int i=0; i<v1.size; i++) {
            if (v2.data[i] == 0) return NULL;   // div by zero
            result->data[i] = v1.data[i] / v2.data[i];
        } break;
        default: return NULL;    // invalid op
    }
    return result;
}
void printVector(const Vector v, FILE *stream) {
    const double *p = v.data;
    for (int i=0; i<v.size; i++) fprintf(stream, "%.4f ", *p++);
    fprintf(stream, "\n");
}

    int tmp = *x;
    *x = *y;
    *y = tmp;

void printBytes(const int *vec, int n) {
    const int *p = vec;
    for (int i = 0; i < n; i++) {
        const unsigned char *b = (const unsigned char *)p++;
        for (int j = 0; j < (int)sizeof(int); j++)
            printf("%hhu ", *b++);
        printf(" ");   // extra space after word
    }
    printf("\n");
}

Recursion — always: base case first (Lab 8)
Recursive printVector (Lab 8)
void printVector(const int *vec, int n) {
    if (n == 0) { printf("\n"); return; } // base
    printf("%d ", *vec);
    printVector(vec + 1, n - 1);           // recurse
}
Recursive reverseVector (Lab 8)
void reverseVector(int *vec, int p1, int p2) {
    if (p1 >= p2) return;       // base case
    int tmp  = vec[p1];
    vec[p1]  = vec[p2];
    vec[p2]  = tmp;
    reverseVector(vec, p1 + 1, p2 - 1);
}
// Call: reverseVector(arr, 0, N-1);
Recursive factorial
int factorial(int n) {
    if (n <= 1) return 1;   // base
    return n * factorial(n - 1);
}
Bit manipulation / binary print (Week 11)
void printBinary(int num) {
    char buf[33];
    buf[32] = '\0';
    for (int i = 31; i >= 0; i--) {
        buf[i] = (num & 1) ? '1' : '0';
        num >>= 1;   // right-shift 1 bit
    }
    printf("%s\n", buf);
}

Stack — array-based (fixed size)
Full array stack (push / pop / peek / print)
#define MAX 100

typedef struct {
    int array[MAX];
    int top;        // -1 = empty
} Stack;

void init(Stack *s)  { s->top = -1; }

void push(Stack *s, int a) {
    if (s->top == MAX - 1) { fprintf(stderr, "Stack full\n"); return; }
    s->array[++s->top] = a;
}

int pop(Stack *s) {
    if (s->top == -1) { fprintf(stderr, "Stack empty\n"); return -1; }
    return s->array[s->top--];
}

int peek(Stack *s) {
    if (s->top == -1) { fprintf(stderr, "Stack empty\n"); return -1; }
    return s->array[s->top];
}

void printStack(Stack *s) {
    for (int i = 0; i <= s->top; i++) printf("%d ", s->array[i]);
    printf("\n");
}

// Usage:
Stack s; init(&s);
push(&s, 5); push(&s, 7);
pop(&s);    peek(&s);
Stack — linked list (dynamic, no size limit)
Linked-list stack (Week 12 / linked list practice)
#include <stdbool.h>

typedef struct node {
    int         item;
    struct node *next;
} Node;

typedef struct {
    int   size;
    Node *top;       // NULL = empty
} Stack;

// push — returns true on success
bool push(Stack *s, int val) {
    Node *n = (Node *) malloc(sizeof(Node));
    if (n == NULL) return false;
    n->item = val;
    n->next = s->top;   // new node points at old top
    s->top  = n;        // new node IS the new top
    s->size++;
    return true;
}

// pop — returns true on success, value via *out
bool pop(Stack *s, int *out) {
    if (s->top == NULL) return false;
    Node *tmp = s->top;
    *out    = tmp->item;
    s->top  = tmp->next;
    s->size--;
    free(tmp);
    return true;
}

void printStack(Stack *s) {
    Node *cur = s->top;
    while (cur != NULL) { printf("%d\n", cur->item); cur = cur->next; }
}

// Usage:
Stack s = {0, NULL};
push(&s, 5);
int v; pop(&s, &v);

queue.h — types and declarations
typedef int ItemType;
#define ITEM_FORMAT "%d"
#define ITEM_PROMPT "an integer"

typedef struct listNode {
    struct listNode *next;
    void            *dataPtr;   // generic pointer
} ListNode;

typedef struct {
    ListNode *front;
    ListNode *rear;
    int       size;
} Queue;

void *enqueue (Queue *q, void *item);
void *dequeue (Queue *q);
int   queueSize(const Queue q);
void  printQueue(const Queue q, FILE *stream);
queue.c — enqueue / dequeue implementations
void *enqueue(Queue *q, void *item) {
    ListNode *node = (ListNode *) malloc(sizeof(ListNode));
    if (node == NULL) return NULL;
    node->next    = NULL;
    node->dataPtr = item;
    if (q->size == 0) {
        q->front = q->rear = node;          // first item
    } else {
        q->rear->next = node; q->rear = node; // add to back
    }
    q->size++;
    return item;
}

void *dequeue(Queue *q) {
    if (q->size == 0) return NULL;
    ListNode *old  = q->front;
    void     *item = old->dataPtr;
    q->front = old->next;
    q->size--;
    if (q->size == 0) q->rear = NULL;   // queue now empty
    free(old);
    return item;
}

void printQueue(const Queue q, FILE *stream) {
    const ListNode *p = q.front;
    while (p != NULL) {
        fprintf(stream, ITEM_FORMAT "\n", *(ItemType *)p->dataPtr);
        p = p->next;
    }
}

void printVector(const WORD *vec, int n) {   // recursive
    if (n == 0) { printf("\n"); return; }
    printf(WORD_FORMAT, *vec);
    printVector(vec+1, n-1);
}
void reverseVector(WORD *vec, int p1, int p2) {
    if (p1 >= p2) return;
    WORD tmp=vec[p1]; vec[p1]=vec[p2]; vec[p2]=tmp;
    reverseVector(vec, p1+1, p2-1);
}
